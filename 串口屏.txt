function on_init()
	local Bt_data = uart_get_baudrate()
    -- 如果返回值为空或为0，则默认设置为9600
    if Bt_data == nil or Bt_data == 0 then
        uart_set_baudrate(9600)
        Bt_data = 9600
    end
end

function on_uart_recv_data(packet, port)
    local hex_str = ""        -- 用于拼接16进制显示
    local data_bytes = {}     -- 存放原始字节数据（0-255）
    local leng = 0
    local start_idx = 0

    -- 自动检测下标起点（0 或 1）
    if packet[1] ~= nil and packet[0] == nil then
        start_idx = 1
    end

    -- 手动统计真实长度并转为16进制显示
    local i = start_idx
    while packet[i] ~= nil do
        leng = leng + 1
        local byte = packet[i]
        table.insert(data_bytes, byte)
        hex_str = hex_str .. string.format("%02X ", byte)
        i = i + 1
    end

    -- 去掉尾部空格
    hex_str = hex_str:gsub("%s+$", "")

    -- 显示收到的16进制数据
    set_text(0, 7, hex_str)

    --------------------------------------------------
    -- 🚀 协议解析逻辑：判断是否温度帧并解析温度
    --------------------------------------------------
    if leng == 8 then
        -- 判断前6字节是否匹配EE 10 00 01 00 01
        if  data_bytes[1] == 0xEE and
            data_bytes[2] == 0x10 and
            data_bytes[3] == 0x00 and
            data_bytes[4] == 0x01 and
            data_bytes[5] == 0x00 and
            data_bytes[6] == 0x01 then

            -- 取出最后两个字节
            local high = data_bytes[7]
            local low  = data_bytes[8]

            -- 组合为16位数
            local raw = high * 256 + low

            -- 处理补码（负数）
            if raw >= 0x8000 then
                raw = raw - 0x10000
            end

            -- 转换为温度，保留一位小数
            local temp = raw / 10.0

            -- 显示温度值
            set_text(2, 10, string.format("%.1f°C", temp))  --原先0，8；现使用新屏幕2，10
        end
    elseif leng == 6 then
        -- 判断是否为 01 02 01 开头的控制指令
        if  data_bytes[1] == 0x01 and
            data_bytes[2] == 0x02 and
            data_bytes[3] == 0x01 then

            -- 取出第4个字节
            local status_byte = data_bytes[4]

            -- 解析bit0（最低位）：控制控件1背景色
            if (status_byte & 0x01) == 0x01 then
                set_back_color(0, 3, 0x07E0)  -- 绿色3
            
            else
                set_back_color(0, 3, 0xF800)  -- 红色3
            
            end

            -- 解析bit1（倒数第二位）：控制控件2背景色
            if (status_byte & 0x02) == 0x02 then
                set_back_color(0, 2, 0xF800)  -- 红色2
            
            else
                set_back_color(0, 2, 0x07E0)  -- 绿色2
            
            end
        end
    else
        set_text(0, 8, "接收长度错误: " .. tostring(leng))
    end    
end
